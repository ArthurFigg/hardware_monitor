# Identificação embutida no .exe, lida pelo Windows.
#
# Não é enfeite: executável anônimo é marcado por heurística de antivírus, e este app já
# soma dois sinais de risco (arquivo único que se descompacta + escrita na chave Run).
# Isto também é o que aparece em Propriedades > Detalhes para quem clica com o botão
# direito antes de executar.
#
# Ao mudar a versão, mudar em TRÊS lugares: filevers e prodvers (tuplas) aqui, as strings
# aqui, e `version` no pyproject.toml. Número que só bate em dois deles vira contradição
# entre o que o arquivo informa nas propriedades e o que o repositório declara.

VSVersionInfo(
  ffi=FixedFileInfo(
    filevers=(2, 1, 0, 0),
    prodvers=(2, 1, 0, 0),
    mask=0x3F,
    flags=0x0,
    OS=0x40004,
    fileType=0x1,
    subtype=0x0,
    date=(0, 0),
  ),
  kids=[
    StringFileInfo(
      [
        StringTable(
          "041604B0",
          [
            StringStruct("CompanyName", "ArthurFigg"),
            StringStruct("FileDescription", "Monitor de Hardware Minimalista"),
            StringStruct("FileVersion", "2.1.0"),
            StringStruct("InternalName", "MonitorDeHardware"),
            StringStruct("LegalCopyright", "© 2026 ArthurFigg"),
            StringStruct("OriginalFilename", "MonitorDeHardware.exe"),
            StringStruct("ProductName", "Monitor de Hardware"),
            StringStruct("ProductVersion", "2.1.0"),
          ],
        )
      ]
    ),
    # 0x0416 = português do Brasil, 0x04B0 = Unicode. Precisa bater com o código da
    # StringTable acima, senão o Windows ignora as strings em silêncio.
    VarFileInfo([VarStruct("Translation", [0x0416, 0x04B0])]),
  ],
)
